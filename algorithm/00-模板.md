/*
 * ============================================================
 * 题号：0001
 * 题目：两数之和
 * 难度：Easy
 * 链接：https://leetcode.cn/problems/two-sum/
 * 题型：01-数组   |   日期：2026-09-18
 * ============================================================
 *
 * 【题目描述】
 * 给定一个整数数组 nums 和一个目标值 target，找出和为目标值的两个整数，返回它们的下标。
 * 示例：nums = [2,7,11,15], target = 9  →  [0,1]
 *
 * ------------------------------------------------------------
 * 【我的思路】（先用大白话写，再写代码！写不出来说明还没懂）
 *
 * 想法一：暴力双层循环，每两个数都试一遍。
 *        时间复杂度 O(n²)。能过但太慢，面试官会问"能优化吗"。
 *
 * 想法二：用 HashMap 把"见过的数"存起来。
 *        遍历到 nums[i] 时，先查 target - nums[i] 在不在 map 里，
 *        在的话就直接返回；不在就把 nums[i] 存进去。
 *        这样只需要遍历一次 → O(n)。
 *
 * 为什么想到用哈希表：因为我要"快速判断某个值之前出现过没有"，
 * 这正是哈希表擅长的事（O(1) 查找）。
 *
 * ------------------------------------------------------------
 * 【复杂度】
 * 时间：O(n)   —— 只遍历一次数组
 * 空间：O(n)   —— 最坏情况 map 里存 n 个数
 *
 * 【我踩的坑】
 * - 一开始把 put 写在了查找前面，导致 [3,3] target=6 时返回 [0,0]。
 *   正确顺序：先查，再存。
 *
 * 【面试可能追问】
 * - 如果数组是有序的呢？（→ 双指针，O(n) 时间 O(1) 空间）
 * - 如果要求返回所有组合呢？（→ 不能提前 return，要遍历完）
 * - 如果允许用同一个元素两次呢？
 * ============================================================
 */

import java.util.Arrays;
import java.util.HashMap;
import java.util.Map;

public class Solution {

    /**
     * 解法一：暴力枚举
     * 面试时先说出这个思路，再给优化版 —— 展示你的思考过程。
     */
    public int[] twoSumBruteForce(int[] nums, int target) {
        for (int i = 0; i < nums.length; i++) {
            for (int j = i + 1; j < nums.length; j++) {
                if (nums[i] + nums[j] == target) {
                    return new int[]{i, j};
                }
            }
        }
        return new int[]{};
    }

    /**
     * 解法二：哈希表（推荐）
     * 关键点：先查后存，避免自己和自己配对。
     */
    public int[] twoSum(int[] nums, int target) {
        // key = 数值, value = 下标
        Map<Integer, Integer> seen = new HashMap<>(nums.length);

        for (int i = 0; i < nums.length; i++) {
            int need = target - nums[i];

            // 先查：之前见过 need 吗？
            if (seen.containsKey(need)) {
                return new int[]{seen.get(need), i};
            }

            // 再存：把当前数记下来，给后面的人用
            seen.put(nums[i], i);
        }

        return new int[]{};
    }

    // ---- 本地自测（不用提交到 LeetCode 也能跑） ----
    public static void main(String[] args) {
        Solution s = new Solution();

        check(Arrays.toString(s.twoSum(new int[]{2, 7, 11, 15}, 9)), "[0, 1]", "普通情况");
        check(Arrays.toString(s.twoSum(new int[]{3, 3}, 6)), "[0, 1]", "重复元素（易错！）");
        check(Arrays.toString(s.twoSum(new int[]{3, 2, 4}, 6)), "[1, 2]", "答案不在开头");
    }

    private static void check(String actual, String expected, String name) {
        boolean ok = actual.equals(expected);
        System.out.printf("%s %-20s 期望=%s 实际=%s%n",
                ok ? "✅" : "❌", name, expected, actual);
    }
}
